#include <Arduino.h>
#include <WiFi.h>

#include "display.h"
#include "api_client.h"
#include "wifi_manager.h"

// fetch every 5 seconds
static const unsigned long FETCH_INTERVAL_MS = 5000;

static unsigned long lastFetch = 0;

void setup() {

  Serial.begin(115200);
  delay(100);

  setupDisplay();

  // show 0 while booting
  setNumber(0);

  Serial.println();
  Serial.println("================================");
  Serial.println("WEBBI STARTING");
  Serial.println("================================");

  connectWiFi();

  // first fetch immediately
  fetchVisitorCount();

  setNumber(currentVisitors);

  lastFetch = millis();
}

void loop() {

  // ALWAYS refresh display
  refreshDisplay();

  unsigned long now = millis();

  if (now - lastFetch >= FETCH_INTERVAL_MS) {

    lastFetch = now;

    Serial.println();
    Serial.println("Fetching visitor count...");

    if (WiFi.status() != WL_CONNECTED) {

      Serial.println("WiFi disconnected, reconnecting...");

      WiFi.reconnect();

      return;
    }

    fetchVisitorCount();

    Serial.print("Visitors: ");
    Serial.println(currentVisitors);

    setNumber(currentVisitors);
  }
}
