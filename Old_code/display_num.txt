int segPins[7] = {18, 19, 21, 22, 23, 25, 26};

// Digit pins: D1 D2 D3 D4 D5 D6
int digitPins[6] = {4, 5, 12, 13, 14, 27};


//HOW DO I PROPERLY SET THE STUFF UP AND HOW DO I PROPERLY INITIALIZE WITH ESP32:

// TOS-11060BE-B:
    // Common Anode
    // 4v per segment (2x 2v Red Led's)
void setup()  {
    for (int i = 0; i < 7; i++) {
        pinMode(segPins[i], OUTPUT);
        digitalWrite(segPins[i], HIGH); // segments OFF initially (common anode)
    }


  // digit pins
    for (int i = 0; i < 6; i++) {
        pinMode(digitPins[i], OUTPUT);
        digitalWrite(digitPins[i], HIGH);  // digits OFF initially (common anode)
    }

}
