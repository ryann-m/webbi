#include <Arduino.h>

// Segment pins in your current order: A B C D E F G
int segPins[7] = {18, 19, 21, 22, 23, 25, 26};

// From your original array:
// int digPins[6] = { 32, 33, 27, 14, 4, 17 };
// You said the 3rd digit is the one lighting, so that is index 2 -> pin 27
int digitPin = 5;

void setup() {
  // Set all segment pins as outputs
  for (int i = 0; i < 7; i++) {
    pinMode(segPins[i], OUTPUT);
    digitalWrite(segPins[i], HIGH); // OFF for common-anode
  }

  // Set the one digit pin as output
  pinMode(digitPin, OUTPUT);
  digitalWrite(digitPin, HIGH); // digit OFF for common-anode

  // Turn on all 7 segments
  for (int i = 0; i < 7; i++) {
    digitalWrite(segPins[i], LOW); // ON for common-anode
  }

  // Turn on this one digit
  digitalWrite(digitPin, LOW); // ON for common-anode
}

void loop() {
}
